\section{Requisitos Mínimos}

\begin{frame}{Passo 1 — Checar Requisitos Mínimos}
\begin{itemize}
  \item Docker Desktop (modo Linux) usa o \textbf{WSL2}
  \item Pré-requisitos: WSL2, Windows compatível e virtualização habilitada
\end{itemize}
\end{frame}

\begin{frame}{1.1 Verificar Virtualização}
\begin{itemize}
  \item Abra o \textbf{Gerenciador de Tarefas} (Ctrl + Shift + Esc)
  \item Aba \textbf{Desempenho} $\rightarrow$ \textbf{CPU}
  \item Procure: \textbf{Virtualização: Habilitada/Enabled}
  \item Se estiver \textbf{Desabilitada}, ajuste na BIOS/UEFI
\end{itemize}
\end{frame}

\section{Instalar WSL2}

\begin{frame}{Passo 2 — Instalar WSL2}
\begin{itemize}
  \item WSL2 é um Linux integrado no Windows
  \item Necessário para o Docker Desktop funcionar corretamente
\end{itemize}
\end{frame}

\begin{frame}{2.1 Abrir Terminal como Administrador}
\begin{itemize}
  \item Clique com botão direito no \textbf{menu iniciar}
  \item Abra \textbf{Windows PowerShell (Admin)} ou \textbf{Terminal (Admin)}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{2.2 Instalar WSL}
\begin{itemize}
  \item Execute o comando de instalação do WSL
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|wsl --install
\end{bashprompt}

\begin{itemize}
  \item Instala o WSL2 e uma distro padrão (geralmente Ubuntu)
  \item Pode pedir reinício do sistema
\end{itemize}
\end{frame}

\begin{frame}[fragile]{2.3 Atualizar o WSL}
\begin{itemize}
  \item Após reiniciar (se necessário), execute:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|wsl --update
\end{bashprompt}

\begin{itemize}
  \item Docker recomenda WSL 2 recente (ex.: 2.1.5+)
\end{itemize}
\end{frame}

\begin{frame}[fragile]{2.4 Confirmar Versão WSL}
\begin{itemize}
  \item Verifique as distros instaladas e suas versões:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|wsl -l -v
\end{bashprompt}

\begin{itemize}
  \item Lista as distros instaladas no WSL e mostra a versão (1 ou 2)
  \item Se aparecer \textbf{VERSION = 2}, você já tem o Linux base pronto
  \item Se não houver distribuições, instale uma distro (próximo passo)
\end{itemize}
\end{frame}

\begin{itemize}
  \item Veja quais distros podem ser instaladas:

\begin{bashprompt}

\begin{itemize}
  \item Mostra distros que o Windows pode baixar e instalar automaticamente
  \item Procure por \textbf{Ubuntu-22.04} (recomendado) ou \textbf{Ubuntu}
\end{itemize}

\begin{frame}[fragile]{Instalar Ubuntu}
\begin{itemize}
  \item Se tiver Ubuntu-22.04 disponível:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|wsl --install -d Ubuntu-22.04
\end{bashprompt}

\begin{itemize}
  \item Baixa e instala a distro Linux dentro do WSL
  \item Durante a primeira execução, o Ubuntu pedirá:
  \begin{itemize}
    \item Um nome de usuário Linux
    \item Uma senha
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Configuração do Usuário}
\begin{itemize}
  \item Você verá: \texttt{Enter new UNIX username:}
  \item Digite um nome simples (sem espaços)
  \item Depois ele pede senha
  \item Digite a senha e aperte Enter
  \item Digite de novo para confirmar
  \item \textbf{Importante}: quando você digita a senha no Linux, não aparece nada na tela (nem asteriscos)
  \item Isso não é bug; é intencional
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Ubuntu Pronto}
\begin{itemize}
  \item Quando terminar, você verá um prompt como:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|username@DESKTOP-XXXXX:~$
\end{bashprompt}

\begin{itemize}
  \item Isso significa que o Ubuntu está pronto e você está dentro dele
\end{itemize}
\end{frame}

\section{Configurar Ubuntu}

\begin{frame}[fragile]{Atualizar o Ubuntu}
\begin{itemize}
  \item Ainda dentro do terminal do Ubuntu, execute:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|sudo apt update
\end{bashprompt}

\begin{itemize}
  \item \textbf{sudo}: executa como administrador do Linux (pode pedir sua senha)
  \item \textbf{apt update}: atualiza a lista de pacotes disponíveis
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Instalar Atualizações}
\begin{itemize}
  \item Agora execute:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|sudo apt upgrade -y
\end{bashprompt}

\begin{itemize}
  \item \textbf{apt upgrade}: instala atualizações dos pacotes já instalados
  \item \textbf{-y}: responde "sim" automaticamente para perguntas
  \item Se aparecer tela de configuração, na dúvida mantenha a versão atual (keep)
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Confirmar Instalação no PowerShell}
\begin{itemize}
  \item Volte ao PowerShell e execute:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|wsl -l -v
\end{bashprompt}

\begin{itemize}
  \item Você deve ver: Ubuntu 22.04 listado, e uma coluna VERSION
  \item Confirme que está em WSL2
\end{itemize}
\end{frame}

\section{Configurar Docker Desktop}

\begin{frame}{Configurar Docker Desktop}
\begin{itemize}
  \item Abra o Docker Desktop (Menu Iniciar)
  \item Vá em \textbf{Settings} e configure:
  \begin{itemize}
    \item \textbf{General}: habilitado "Use the WSL 2 based engine"
    \item \textbf{Resources} $\rightarrow$ \textbf{WSL Integration}: habilite a integração com Ubuntu-22.04
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{O que faz a WSL Integration}
\begin{itemize}
  \item O "motor" do Docker fica rodando em uma distro interna dele
  \item A opção "WSL Integration" permite você digitar \texttt{docker ...} dentro do Ubuntu (WSL) e isso funcionar
  \item Sem isso, o Docker não funciona dentro do WSL
\end{itemize}
\end{frame}

\section{ROS 2 Humble em Docker}

\begin{frame}[fragile]{Baixar e Entrar no ROS 2 Humble}
\begin{itemize}
  \item A documentação do ROS2 sugere usar a imagem \texttt{osrf/ros:humble-desktop}
  \item No Ubuntu (WSL), baixe a imagem:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|docker pull osrf/ros:humble-desktop
\end{bashprompt}
\end{frame}

\begin{frame}[fragile]{Entrar no Container ROS 2}
\begin{itemize}
  \item Depois entre no container:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|docker run -it --name ros2-humble osrf/ros:humble-desktop
\end{bashprompt}

\begin{itemize}
  \item Você entra num "Linux isolado" que já tem ROS 2 Humble instalado
  \item Para sair do container depois: \texttt{exit}
  \item Para voltar a entrar no mesmo container (sem recriar):
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|docker start -ai ros2-humble
\end{bashprompt}
\end{frame}

\section{ROS2 Tutorial}

\begin{frame}{Configurar Ambiente ROS 2}
\begin{itemize}
  \item Sem "sourcing" do setup, o terminal não "enxerga" os comandos do ROS2
  \item Dentro do container, configure o ambiente
\end{itemize}
\end{frame}

\begin{frame}[fragile]{4.1 Ligar o ROS2 no Terminal Atual}
\begin{itemize}
  \item Dentro do container, execute:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|source /opt/ros/humble/setup.bash
\end{bashprompt}

\begin{itemize}
  \item Isso ativa os comandos do ROS2 neste terminal
\end{itemize}
\end{frame}

\begin{frame}[fragile]{4.2 Fazer isso Automaticamente Sempre}
\begin{itemize}
  \item Para configurar automaticamente em cada abertura:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc
\end{bashprompt}

\begin{itemize}
  \item Isso coloca o comando no "arquivo de inicialização" do terminal do container
  \item O ROS2 será carregado automaticamente ao entrar no container
\end{itemize}
\end{frame}

\begin{frame}[fragile]{4.3 Conferir Variáveis do ROS}
\begin{itemize}
  \item Verifique se as variáveis estão configuradas:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|printenv | grep -i ROS
\end{bashprompt}

\begin{itemize}
  \item Você deve ver: \texttt{ROS\_DISTRO=humble}, \texttt{ROS\_VERSION=2}
\end{itemize}
\end{frame}

\section{Tutorial Turtlesim}

\begin{frame}[fragile]{Como Abrir Vários Terminais}
\begin{itemize}
  \item O tutorial pede 2–3 terminais ao mesmo tempo
  \item No Docker, a forma simples é:
  \begin{itemize}
    \item \textbf{Terminal A}: é onde você entrou no container ros2-humble
    \item \textbf{Terminal B e C}: abra novas abas do Ubuntu (WSL) e rode:
  \end{itemize}
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|docker exec -it ros2-humble bash
\end{bashprompt}

\begin{itemize}
  \item Isso abre um "novo terminal" dentro do mesmo container
  \item Regra prática: em cada terminal novo dentro do container, rode:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|source /opt/ros/humble/setup.bash
\end{bashprompt}
\end{frame}

\section{Remapping}

\begin{frame}[fragile]{5) Remapping: Controlar a turtle2}
\begin{itemize}
  \item A turtle2 não tem teleop por padrão
  \item O tutorial resolve isso "remapeando" o tópico de velocidade
  \item Abra um Terminal D (ou reutilize um terminal livre) e rode:
\end{itemize}

\begin{bashprompt}
|\nocopy{\textdollar\ }|source /opt/ros/humble/setup.bash
|\nocopy{\textdollar\ }|ros2 run turtlesim turtle_teleop_key --ros-args --remap turtle1/cmd_vel:=turtle2/cmd_vel
\end{bashprompt}

\begin{itemize}
  \item Agora:
  \begin{itemize}
    \item Quando esse terminal estiver "ativo", as setas controlam a turtle2
    \item O outro teleop continua controlando a turtle1
  \end{itemize}
\end{itemize}
\end{frame}
