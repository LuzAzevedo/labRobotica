    def __init__(self):
        super().__init__('qos_demo')
        
        # Publisher com Transient Local (Ex: Status do Robô)
        qos_profile = QoSProfile(
            depth=1,
            durability=QoSDurabilityPolicy.TRANSIENT_LOCAL
        )
        
        self.pub = self.create_publisher(
            String, 'robot_state', qos_profile)
            
        # Envia uma vez, quem conectar depois recebe!
        msg = String(data="MODO_MANUTENCAO")
        self.pub.publish(msg)
