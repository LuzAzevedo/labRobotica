from launch_ros.actions import Node

def generate_launch_description():
    return LaunchDescription([
        # Nó 1: Publisher (com remap)
        Node(
            package='example_interfaces',
            executable='one_off_timer_control', # Apenas exemplo
            remappings=[('/one_off_timer_control', '/cmd_start')]
        ),
        # Nó 2: Turtlesim (com namespace)
        Node(
            package='turtlesim',
            executable='turtlesim_node',
            namespace='turtles',
            parameters=[{'background_b': 200}]
        )
    ])
